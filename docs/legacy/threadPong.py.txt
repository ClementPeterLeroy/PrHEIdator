def server_Flask():
    def run():
        app = Flask(__name__)
        global nbPlayers
        global currentPlayer
        global LidManette
        global Lpseudo
        global idSession

        @app.route('/')
        def hllo():
            return Response("Hello !", 200)

        @app.route('/register', methods=['GET'])
        def register():
            global currentPlayer
            global LidManette
            global idSession
            if (nbPlayers == currentPlayer):
                print(f"nbPlayers : {nbPlayers}")
                print(f"current : {currentPlayer}")
                return Response("Session Full", 403)
            else:
                idManette = request.args.get("idmanette")
                if (idManette in LidManette):
                    pos = LidManette.index(idManette)
                    pseudo = request.args.get("pseudo")
                    check = Parsing.checkPseudo(pseudo)
                    if (check == False):
                        return Response("Invalid Pseudo", 400)
                    else:
                        Lpseudo[pos] = pseudo
                        return Response(str(idSession), 201)
                else:
                    pseudo = request.args.get("pseudo")
                    check = Parsing.checkPseudo(pseudo)
                    if (check == False):
                        return Response("Invalid Pseudo", 400)
                    else:
                        print("coucou")
                        Lpseudo.append(pseudo)
                        LidManette.append(idManette)
                        print(f"current : {currentPlayer}")
                        currentPlayer += 1
                        print(f"nbPlayers : {nbPlayers}")
                        print(f"current : {currentPlayer}")
                        return Response(str(idSession), 200)
        app.run()

    t2 = threading.Thread(target=run)
    t2.start()
    print("Flask Started")

    def consume():

        global LidManette

        topic='keys'
        consumer = KafkaConsumer(topic, bootstrap_servers=BOOTSTRAP_SERVERS)

        print("About to start polling for topic:", topic)
        consumer.poll(timeout_ms=1000)
        print("Started Polling for topic:", topic)
        for msg in consumer:

            J = json.loads(x.decode("utf-8").replace("'", '"'))
